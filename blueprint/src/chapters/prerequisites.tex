\chapter{Prerequisites}
\label{chap:prerequisites}

The library-shaped results that the paper needs and that LMLExplore, LML, Tau Ceti and Mathlib do
not provide yet. Their Lean code goes to the library's \texttt{ForLMLExplore/} folder, and
\texttt{PAPER\_ROADMAP.md} says where each goes in LMLExplore.

\begin{theorem}
  \label{prereq:scalar}
  For arms in an exponential family and a vector of means with unique best arm $a$, the optimal
  allocation is $x / \sum_c x_c$, with $x_a = 1$ and $x_b = g_b^{-1}(y^*)$ for $b \ne a$, where
  $g_b(x) = d(\mu_a, m_b(x)) + x\, d(\mu_b, m_b(x))$ is a strictly increasing bijection from
  $[0, \infty)$ onto $[0, d(\mu_a, \mu_b))$ and $y^*$ is the unique solution of $F(y) = 1$, for the
  continuous increasing function
  $F(y) = \sum_{b \ne a} d(\mu_a, m_b(x_b(y))) / d(\mu_b, m_b(x_b(y)))$ with $F(0) = 0$ and
  $F(y) \to \infty$ as $y \to \min_{b \ne a} d(\mu_a, \mu_b)$ \cite[Theorem 5]{GK16}.
  LMLRoadmaps: Identification 3.4.
\end{theorem}

\begin{lemma}
  \label{prereq:pair_cost_anti}
  \lean{MeasureTheory.Measure.strictAntiOn_expFamilyPairCost_right,
    MeasureTheory.Measure.antitoneOn_expFamilyPairCost_right}
  \leanok
  For an exponential family with divergence $d$, a mean $\mu_1$ of the family and nonnegative
  weights $w_1, w_2$, the cost $w_1 d(\mu_1, m) + w_2 d(\mu_2, m)$ of moving $\mu_1$ and $\mu_2$ to
  their weighted mean $m = (w_1\mu_1 + w_2\mu_2)/(w_1 + w_2)$ is nonincreasing in the mean
  $\mu_2 \le \mu_1$ of the family. It is decreasing when $w_1, w_2 > 0$.
\end{lemma}

\begin{proof}
  \leanok
  Let $\mu_2' < \mu_2 \le \mu_1$ and let $m'$ be the weighted mean of $\mu_1$ and $\mu_2'$, so that
  $\mu_2' < m' \le \mu_1$. The pair cost of $(\mu_1, \mu_2)$ is the minimum of
  $w_1 d(\mu_1, x) + w_2 d(\mu_2, y)$ over the means $x \le y$. If $\mu_2 \le m'$, take
  $x = y = m'$: then $d(\mu_2, m') < d(\mu_2', m')$, since $d(\cdot, m')$ decreases below $m'$. If
  $m' < \mu_2$, take $x = m'$ and $y = \mu_2$: then $d(\mu_2, \mu_2) = 0 < d(\mu_2', m')$. In both
  cases the result is at most $w_1 d(\mu_1, m') + w_2 d(\mu_2', m')$, the pair cost of
  $(\mu_1, \mu_2')$, and strictly less when $w_2 > 0$. If a weight vanishes, both costs are zero.
\end{proof}

\begin{theorem}
  \label{prereq:ordering}
  \lean{LMLExplore.Identification.le_iff_le_of_mem_optAlloc_baiTask_restrict,
    LMLExplore.Identification.lt_iff_lt_of_mem_optAlloc_baiTask_restrict,
    LMLExplore.Identification.le_of_mem_betaOptAlloc_baiTask_restrict_of_le,
    LMLExplore.Identification.lt_of_mem_betaOptAlloc_baiTask_restrict_of_lt}
  \leanok
  \uses{lem:equalize, prereq:pair_cost_anti}
  For arms in an exponential family and a vector of means with unique best arm $a$, the optimal
  allocation is ordered like the means on the arms other than $a$: for $b, c \ne a$,
  $w^*_c(\mu) \le w^*_b(\mu)$ if and only if $\mu_c \le \mu_b$, and $w^*_c(\mu) < w^*_b(\mu)$ if
  and only if $\mu_c < \mu_b$ \cite[Proposition 6]{GK16}. The same holds for the $\beta$-optimal
  allocations, the maximizers of $D(\mu, w, \neg a)$ among the allocations with $w_a = \beta$: for
  $\beta > 0$, and in the strict form for $\beta \in (0, 1)$. No LMLRoadmaps target covers it yet.
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:equalize, prereq:pair_cost_anti}
  Let $w$ be $\beta$-optimal, so that the costs $C_b(w_b)$ of the pairs $(a, b)$, with
  $C_b(x) = \beta d(\mu_a, m_b) + x d(\mu_b, m_b)$, are all equal (\cref{lem:equalize} and its
  $\beta$ form). Let $b, c \ne a$ with $\mu_c \le \mu_b$. If $w_b < w_c$, then $C_b$ increases, so
  $C_b(w_b) < C_b(w_c)$. By \cref{prereq:pair_cost_anti}, $C_b(w_c) \le C_c(w_c)$. This
  contradicts $C_b(w_b) = C_c(w_c)$. If $\mu_c < \mu_b$ and $w_b \le w_c$, then
  $C_b(w_b) \le C_b(w_c) < C_c(w_c)$, using $w_c > 0$, a contradiction again. The optimal
  allocation is $\beta$-optimal for $\beta = w^*_a \in (0, 1)$; the two implications give the
  equivalences.
\end{proof}

\begin{lemma}
  \label{prereq:kt}
  The Krichevsky--Trofimov distribution
  $\mathrm{kt}(x) = \int_0^1 \frac{1}{\pi\sqrt{u(1-u)}} p_u(x)\, du$ is a probability on
  $\{0, 1\}^n$ with $\sup_x \sup_u p_u(x) / \mathrm{kt}(x) \le 2\sqrt n$ \cite[Lemma 11]{GK16},
  and with Bernoulli arms the GLRT with the threshold $\log(2t(K-1)/\delta)$ is $\delta$-correct
  whatever the sampling rule \cite[Theorem 10]{GK16}. LMLRoadmaps: Identification 4.2.
\end{lemma}

\begin{lemma}
  \label{prereq:iid_slice}
  \lean{ProbabilityTheory.iIndepFun.measure_exists_le_and_gt_and_lt_rateFunction_le,
    ProbabilityTheory.iIndepFun.measure_exists_le_and_lt_and_le_mul_rateFunction_le}
  \leanok
  Let $Z_0, Z_1, \dots$ be i.i.d.\ with the law of an integrable $Y$, with rate function $I$, and
  let $\hat\mu_n$ be the mean of $Z_0, \dots, Z_{n-1}$. For $a > 0$ and $r \ge 0$,
  $\Prob(\exists n \ge a,\ \hat\mu_n > \E[Y],\ I(\hat\mu_n) > r) \le e^{-ar}$. For $a > 0$ and
  $z > 0$, $\Prob(\exists n,\ a \le n < b,\ n I(\hat\mu_n) \ge z) \le 2 e^{-az/b}$.
\end{lemma}

\begin{proof}
  \leanok
  Along the filtration of the past variables, $Z_n$ is independent of the past, so its
  conditional moment generating function is that of $Y$, and the first bound is LMLExplore's
  time-uniform Chernoff bound with the rate function, with every variable selected. For the
  second, $n I(\hat\mu_n) \ge z$ with $n < b$ forces $I(\hat\mu_n) > z/b$, which vanishes at the
  mean; above the mean this is the first bound with $r = z/b$, and below the mean the first bound
  for $-Z$ and $-Y$.
\end{proof}

\begin{lemma}
  \label{prereq:peeled_arm}
  \lean{LMLExplore.streamMeasure_exists_two_mul_add_le_mul_rateFunction_le}
  \leanok
  \uses{prereq:iid_slice}
  In the reward stream model, for an arm with integrable rewards and rate function $I_a$, every
  $\eta > 0$ and $y \ge 0$, with $\kappa_\eta = \log(2/(1 - e^{-\eta}))$,
  $\Prob(\exists s,\ s I_a(\hat\mu_s) \ge 2(\eta \lfloor\log_2 s\rfloor + \kappa_\eta + y))
  \le e^{-y}$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{prereq:iid_slice}
  A deviation with $s \ge 1$ rewards lies on the dyadic slice $2^k \le s < 2^{k+1}$ with
  $k = \lfloor\log_2 s\rfloor$, and at $s = 0$ there is none. By \cref{prereq:iid_slice} with
  $a = 2^k$ and $b = 2^{k+1}$, the slice $k$ has probability at most
  $2 e^{-(\eta k + \kappa_\eta + y)}$, and these sum to
  $2 e^{-\kappa_\eta - y} / (1 - e^{-\eta}) = e^{-y}$.
\end{proof}

\begin{lemma}
  \label{prereq:peeled_joint}
  \lean{LMLExplore.streamMeasure_exists_two_mul_card_mul_add_le_sum_mul_rateFunction_le,
    LMLExplore.measure_exists_two_mul_card_mul_add_le_sum_pullCount_mul_rateFunction_le,
    LMLExplore.measure_exists_two_mul_card_mul_add_le_sum_pullCount_mul_expFamilyRate_le}
  \leanok
  \uses{prereq:peeled_arm}
  Along a run of any algorithm, if the rewards of each arm are integrable with rate function
  $I_a$, then for $\eta > 0$ and $x \ge K$,
  $\Prob(\exists t,\ \sum_a N_a(t) I_a(\hat\mu_a(t)) \ge
  2(K(\eta \lfloor\log_2 t\rfloor + \kappa_\eta) + x)) \le (x/K)^K e^{K - x}$. For arms dominated
  by members of exponential families, the same holds with the extended divergences
  $\bar d_a(\hat\mu_a(t), \mu_a)$ in place of $I_a(\hat\mu_a(t))$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{prereq:peeled_arm}
  In the stream model the arms are independent, and \cref{prereq:peeled_arm} gives each half
  statistic $s I_a(\hat\mu_s)/2$ the exponential tail above the monotone function
  $\eta \lfloor\log_2 s\rfloor + \kappa_\eta$ that \cite[Proposition 33.9]{LS} (in LMLExplore)
  combines over the arms. LMLExplore's joint event transfer turns the stream bound into a bound
  along the run, where $\sum_a N_a(t) = t$. The rate function of a dominated arm is at least the
  extended divergence of its family.
\end{proof}

\begin{lemma}
  \label{prereq:deviational_glrt}
  \lean{LMLExplore.Identification.IdentTask.measure_exists_lt_expFamilyEvidence_le_of_le,
    LMLExplore.Identification.IdentTask.measure_exists_lt_expFamilyEvidence_le_of_mul_log_le,
    LMLExplore.Identification.IdentTask.measureReal_histOutputMeasure_glrt_expFamilyEvidence_le_of_mul_log_le}
  \leanok
  \uses{prereq:peeled_joint}
  For every identification task on vectors of means, arms dominated by exponential families,
  every sampling rule, $p > 0$ and $x \ge K$, the GLRT with the generalized likelihood ratio
  evidence and thresholds at least $p \log n + c_{p,K} + 2x$ at every round $n \ge 1$, with
  $c_{p,K} = 2K \log(2/(1 - e^{-p\log 2/(2K)}))$, stops with an incorrect answer with
  probability at most $(x/K)^K e^{K-x}$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{prereq:peeled_joint}
  The GLRT errs only if the evidence against the true model exceeds the threshold at some round,
  which does not happen at round $0$, where the evidence vanishes. With
  $\eta = p\log 2/(2K)$, $2K\eta\lfloor\log_2 n\rfloor \le p \log n$, so the thresholds dominate
  those of \cref{prereq:peeled_joint}.
\end{proof}

\begin{lemma}
  \label{prereq:deviational}
  \lean{LMLExplore.Identification.IdentTask.measureReal_histOutputMeasure_glrt_expFamilyEvidence_log_mul_rpow_div_le,
    LMLExplore.Identification.IdentTask.isDeltaPAC_glrt_expFamilyEvidence_log_mul_rpow_div}
  \leanok
  \uses{prereq:deviational_glrt}
  For every identification task on vectors of means, arms dominated by exponential families,
  every sampling rule, $p > 0$, $\delta \in (0, 1]$ and $C \ge \delta e^{c_{p,K} + 2x_\delta}$,
  where $x_\delta = \log(1/\delta) + 2K(1 + \log(1 + \log(1/\delta)))$ is LMLExplore's level,
  the GLRT with the threshold $\log(C t^p / \delta)$ is $\delta$-correct
  \cite[Proposition 12]{GK16}, and it is $\delta$-PAC when it stops almost surely.
  LMLRoadmaps: Identification 4.2.
\end{lemma}

\begin{proof}
  \leanok
  \uses{prereq:deviational_glrt}
  For $n \ge 1$, $\log(C n^p/\delta) \ge p \log n + c_{p,K} + 2x_\delta$, and
  $(x_\delta/K)^K e^{K - x_\delta} \le \delta$ with $x_\delta \ge K$ (LMLExplore).
\end{proof}

\begin{lemma}
  \label{prereq:bai_continuity}
  \lean{LMLExplore.Identification.isOpen_answerSet_baiTask_restrict,
    LMLExplore.Identification.continuousWithinAt_altCost_baiTask_restrict}
  \leanok
  For best arm identification restricted to an open set of vectors of means, such as the vectors
  of means of a nondegenerate exponential family, the models with unique best arm $a$ form an open
  set. For arms in a nondegenerate exponential family and best arm identification restricted to
  the vectors of means of the family, at every model $\mu$ with unique best arm $a$,
  $(\mu', w) \mapsto D(\mu', w, \neg a)$ is continuous within $\R^K \times \Sk$ at $(\mu, w)$ for
  every $w \in \Sk$. These are the continuity hypotheses of LMLExplore's sample complexity bounds
  for Track-and-Stop. LMLExplore proves the continuity as a private lemma; LMLExplore pull request
  \#242 makes it public and adds the openness.
\end{lemma}

\begin{proof}
  \leanok
  The models with unique best arm $a$ are an intersection of finitely many strict inequalities
  between means, intersected with the open set of the restriction. The closed form of
  \cref{lem:pair_cost} is continuous on the vectors of means of the family with best arm $a$ and
  the nonnegative allocations, a neighbourhood of $(\mu, w)$ within $\R^K \times \Sk$, since the
  set of means of the family is open.
\end{proof}

\begin{lemma}
  \label{prereq:as_pathwise}
  \lean{LMLExplore.Identification.IdentTask.stoppingTime_glrt_ne_top_of_eventually_mul_le_glr,
    LMLExplore.Identification.IdentTask.eventually_stoppingTime_glrt_le_of_eventually_mul_le_glr}
  \leanok
  Along a sequence of histories on which the GLR statistic of the recommended answer after $t$
  rounds is at least $\kappa t$ for all large $t$, consider the GLRTs with thresholds
  $\beta_t(h, \delta) \le \log(1/\delta) + f(t)$, $f(t) = o(t)$. If $\kappa > 0$, the GLRT stops
  at every $\delta$ at which the thresholds have this form, and $\tau_\delta \le c \log(1/\delta)$
  for all small $\delta$, for every $c > 1/\kappa$.
\end{lemma}

\begin{proof}
  \leanok
  Take $1/c < \eta < \kappa$ and $L = \log(1/\delta)$. From the round $N = \max(M, \lceil L/\eta
  \rceil)$ on, for a fixed $M$, the threshold is at most $L + (\kappa - \eta) N / 2 < \kappa N$, so
  that $\tau_\delta \le N \le M + L/\eta + 1$, which is at most $cL$ when $L$ is large.
\end{proof}

\begin{lemma}
  \label{prereq:glr_growth}
  \lean{LMLExplore.Identification.IdentTask.eventually_mul_le_glr_plugInCost}
  \leanok
  Let $W$ be a compact set of allocations such that $(\theta', w) \mapsto D(\theta', w, \neg i)$ is
  lower semicontinuous at $(\theta, w)$ on the simplex for every $w \in W$, and let the
  recommendation be $i$ near $\theta$. Along a sequence of histories on which the estimates
  converge to $\theta$ and the proportions of pulls $N(t)/t$ approach $W$, the GLR statistic of the
  plug-in statistic, $t D(\hat\theta_t, N(t)/t, \neg i)$, is at least $\kappa t$ for all large $t$,
  for every $\kappa < \inf_{w \in W} D(\theta, w, \neg i)$.
\end{lemma}

\begin{proof}
  \leanok
  By compactness of $W$ and lower semicontinuity, $D(\theta', w', \neg i) > \kappa$ for $\theta'$
  near $\theta$ and $w'$ in the simplex near $W$, which $\hat\theta_t$ and $N(t)/t$ are for all
  large $t$.
\end{proof}

\begin{lemma}
  \label{prereq:as}
  \lean{LMLExplore.Identification.IdentTask.ae_eventually_stoppingTime_glrt_le,
    LMLExplore.Identification.IdentTask.ae_stoppingTime_glrt_ne_top}
  \leanok
  \uses{prereq:as_pathwise, prereq:glr_growth}
  For the GLRT with thresholds $\beta_t(h, \delta) \le \log(1/\delta) + f(t)$, $f(t) = o(t)$, if
  almost surely the estimates converge to the model and the proportions of pulls to a set $W$ of
  allocations, then almost surely $\tau_\delta \le c \log(1/\delta)$ for all small $\delta$, for
  every $c > (\inf_{w \in W} D(\mu, w, \neg a^*))^{-1}$ \cite[Proposition 13]{GK16}. If the
  infimum is positive, the GLRT stops almost surely at every $\delta$ at which the thresholds have
  this form. LMLRoadmaps: Identification 4.4.
\end{lemma}

\begin{proof}
  \leanok
  \uses{prereq:as_pathwise, prereq:glr_growth}
  On the almost sure event of the convergences, \cref{prereq:glr_growth} gives the linear growth
  of the statistic, and \cref{prereq:as_pathwise} the bounds on the stopping time.
\end{proof}

\begin{lemma}
  \label{prereq:slln}
  \lean{LMLExplore.ae_tendsto_empMean_of_tendsto_pullCount}
  \leanok
  In a run of any algorithm against a stochastic bandit with integrable rewards, almost surely the
  empirical mean of every arm pulled infinitely often converges to the mean of the arm.
\end{lemma}

\begin{proof}
  \leanok
  The rewards of the successive pulls of an arm are i.i.d.\ with the law of the arm (in LML), and
  after $N_a(t)$ pulls the empirical mean is the mean of the first $N_a(t)$ of them: the strong law
  of large numbers applies.
\end{proof}
