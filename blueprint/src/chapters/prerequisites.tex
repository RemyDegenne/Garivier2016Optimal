\chapter{Prerequisites}
\label{chap:prerequisites}

The library-shaped results that the paper needs and that LMLExplore, LML, Tau Ceti and Mathlib do
not provide yet. Their Lean code goes to the library's \texttt{ForLMLExplore/} folder, and

\begin{lemma}
  \label{prereq:pair_cost_linear}
  \lean{MeasureTheory.Measure.expFamilyPairCost_le_of_mem_expFamilyMeans}
  \leanok
  For an exponential family with divergence $d$, means $\mu_1, \mu_2$ of the family and
  nonnegative weights $w_1, w_2$, the cost $w_1 d(\mu_1, m) + w_2 d(\mu_2, m)$ of moving $\mu_1$
  and $\mu_2$ to their weighted mean $m$ is at most $w_1 d(\mu_1, z) + w_2 d(\mu_2, z)$ for every
  mean $z$ of the family. So the cost of a pair is a minimum of functions linear in the weights,
  and equal to the one with $z = m$ at the weights $(w_1, w_2)$.
\end{lemma}

\begin{proof}
  \leanok
  If $w_1 = w_2 = 0$, both sides vanish. Otherwise $m$ is a mean of the family, and it minimizes
  $z \mapsto w_1 d(\mu_1, z) + w_2 d(\mu_2, z)$ by the compensation identity
  $w_1 d(\mu_1, z) + w_2 d(\mu_2, z) = w_1 d(\mu_1, m) + w_2 d(\mu_2, m) + (w_1 + w_2) d(m, z)$.
\end{proof}

\begin{lemma}
  \label{prereq:div_mean_continuous}
  \lean{MeasureTheory.Measure.continuousAt_expFamilyDiv_div}
  \leanok
  For means $\mu, \mu_1, \mu_2$ of the family, $(w_1, w_2) \mapsto
  d(\mu, (w_1\mu_1 + w_2\mu_2)/(w_1 + w_2))$ is continuous at every pair of nonnegative weights
  with $w_1 + w_2 > 0$.
\end{lemma}

\begin{proof}
  \leanok
  The weighted mean is continuous there and is a mean of the family, which is open, and $d$ is
  continuous on pairs of means.
\end{proof}

\begin{lemma}
  \label{prereq:scalar_opt}
  \lean{LMLExplore.Identification.mem_optAlloc_baiTask_restrict_of_sum_div_eq_one}
  \leanok
  \uses{lem:pair_cost, prereq:pair_cost_linear}
  For arms in an exponential family and a vector of means $\mu$ with unique best arm $a$, let
  $w \in \Sigma_K$ with $w_a > 0$ equalize the costs
  $C_b(w) = w_a d(\mu_a, m_b) + w_b d(\mu_b, m_b)$ of the pairs $(a, b)$, $b \ne a$, where
  $m_b = (w_a\mu_a + w_b\mu_b)/(w_a + w_b)$, and satisfy
  $F = \sum_{b \ne a} d(\mu_a, m_b) / d(\mu_b, m_b) = 1$. Then $w$ is optimal.
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:pair_cost, prereq:pair_cost_linear}
  Write $A_b = d(\mu_a, m_b)$ and $B_b = d(\mu_b, m_b)$, which is positive since $w_a > 0$ and
  $\mu_b < \mu_a$ give $m_b \ne \mu_b$, and $D$ for the common value of the costs. For every
  $x \in \Sigma_K$, $\sum_{b \ne a} (x_a A_b + x_b B_b) / B_b = x_a F + \sum_{b \ne a} x_b = 1$.
  At $x = w$ this is $\sum_{b \ne a} D / B_b = 1$. Let $v \in \Sigma_K$. By
  \cref{prereq:pair_cost_linear} with $z = m_b$, the cost of the pair $(a, b)$ at $v$ is at most
  $v_a A_b + v_b B_b$. If all of them exceeded $D$, then
  $1 = \sum_{b \ne a} D / B_b < \sum_{b \ne a} (v_a A_b + v_b B_b) / B_b = 1$. So some pair costs
  at most $D$ at $v$, and by \cref{lem:pair_cost} the cost of the alternatives at $v$ is at most
  its value $D$ at $w$.
\end{proof}

\begin{lemma}
  \label{prereq:scalar_eq}
  \lean{LMLExplore.Identification.sum_div_eq_one_of_mem_optAlloc_baiTask_restrict}
  \leanok
  \uses{lem:pair_cost, prereq:pair_cost_linear, prereq:div_mean_continuous}
  For arms in an exponential family, a vector of means $\mu$ with unique best arm $a$ and at
  least one other arm, the optimal allocation satisfies $F = 1$, with the notation of
  \cref{prereq:scalar_opt}.
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:pair_cost, prereq:pair_cost_linear, prereq:div_mean_continuous}
  The optimal allocation $w$ has positive entries (in LMLExplore). Suppose $F \ne 1$, and let
  $S = \sum_{b \ne a} 1 / B_b > 0$. The direction $u$ with $u_a = S / (F - 1)$ and
  $u_b = (1 - A_b u_a) / B_b$ for $b \ne a$ has $\sum_c u_c = u_a + S - F u_a = 0$ and
  $u_a A_b + u_b B_b = 1$ for every $b \ne a$. Let $v = w + t u$, with $m'_b$ the weighted mean
  of $\mu_a$ and $\mu_b$ at $v$. The cost of the pair $(a, b)$ at $v$ is
  $v_a d(\mu_a, m'_b) + v_b d(\mu_b, m'_b)$, and its cost at $w$ is at most
  $w_a d(\mu_a, m'_b) + w_b d(\mu_b, m'_b)$ (\cref{prereq:pair_cost_linear}), so the difference
  is at least $t\,(u_a d(\mu_a, m'_b) + u_b d(\mu_b, m'_b))$. As $t \to 0$, the second factor tends
  to $u_a A_b + u_b B_b = 1$ (\cref{prereq:div_mean_continuous}). So for some small $t > 0$, $v$
  is in the simplex and every pair costs more at $v$ than at $w$: by \cref{lem:pair_cost}, the
  cost of the alternatives is larger at $v$, which contradicts the optimality of $w$. The proof
  needs neither a derivative of $d$ nor the inverses of the costs of the pairs, which the paper
  uses.
\end{proof}

\begin{theorem}
  \label{prereq:scalar}
  \lean{LMLExplore.Identification.mem_optAlloc_baiTask_restrict_iff}
  \leanok
  \uses{lem:equalize, prereq:scalar_opt, prereq:scalar_eq}
  For arms in an exponential family, a vector of means $\mu$ with unique best arm $a$ and at
  least one other arm, an allocation $w$ is optimal if and only if $w \in \Sigma_K$, $w_a > 0$,
  the costs of the pairs $(a, b)$, $b \ne a$, are equal, and
  $\sum_{b \ne a} d(\mu_a, m_b) / d(\mu_b, m_b) = 1$, with the notation of
  \cref{prereq:scalar_opt} \cite[Theorem 5]{GK16}. LMLRoadmaps: Identification 3.4.

  The paper writes the costs of the pairs as $w_a g_b(w_b / w_a)$, with the increasing function
  $g_b(x) = d(\mu_a, m_b(x)) + x\, d(\mu_b, m_b(x))$, and reduces the optimization to the scalar
  equation $F(y) = 1$ in the common value $y$ of the $g_b$, through their inverses. The
  properties of $g_b^{-1}$ and of $F$ as a function of $y$ (monotonicity, limits, the bisection)
  are not formalized: the characterization above does not need them.
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:equalize, prereq:scalar_opt, prereq:scalar_eq}
  The optimal allocation has positive entries (in LMLExplore), equalizes the costs of the pairs
  (\cref{lem:equalize}) and satisfies $F = 1$ (\cref{prereq:scalar_eq}). Conversely,
  \cref{prereq:scalar_opt}.
\end{proof}

\begin{lemma}
  \label{prereq:pair_cost_anti}
  \lean{MeasureTheory.Measure.strictAntiOn_expFamilyPairCost_right,
    MeasureTheory.Measure.antitoneOn_expFamilyPairCost_right}
  \leanok
  For an exponential family with divergence $d$, a mean $\mu_1$ of the family and nonnegative
  weights $w_1, w_2$, the cost $w_1 d(\mu_1, m) + w_2 d(\mu_2, m)$ of moving $\mu_1$ and $\mu_2$ to
  their weighted mean $m = (w_1\mu_1 + w_2\mu_2)/(w_1 + w_2)$ is nonincreasing in the mean
  $\mu_2 \le \mu_1$ of the family. It is decreasing when $w_1, w_2 > 0$.
\end{lemma}

\begin{proof}
  \leanok
  Let $\mu_2' < \mu_2 \le \mu_1$ and let $m'$ be the weighted mean of $\mu_1$ and $\mu_2'$, so that
  $\mu_2' < m' \le \mu_1$. The pair cost of $(\mu_1, \mu_2)$ is the minimum of
  $w_1 d(\mu_1, x) + w_2 d(\mu_2, y)$ over the means $x \le y$. If $\mu_2 \le m'$, take
  $x = y = m'$: then $d(\mu_2, m') < d(\mu_2', m')$, since $d(\cdot, m')$ decreases below $m'$. If
  $m' < \mu_2$, take $x = m'$ and $y = \mu_2$: then $d(\mu_2, \mu_2) = 0 < d(\mu_2', m')$. In both
  cases the result is at most $w_1 d(\mu_1, m') + w_2 d(\mu_2', m')$, the pair cost of
  $(\mu_1, \mu_2')$, and strictly less when $w_2 > 0$. If a weight vanishes, both costs are zero.
\end{proof}

\begin{theorem}
  \label{prereq:ordering}
  \lean{LMLExplore.Identification.le_iff_le_of_mem_optAlloc_baiTask_restrict,
    LMLExplore.Identification.lt_iff_lt_of_mem_optAlloc_baiTask_restrict,
    LMLExplore.Identification.le_of_mem_betaOptAlloc_baiTask_restrict_of_le,
    LMLExplore.Identification.lt_of_mem_betaOptAlloc_baiTask_restrict_of_lt}
  \leanok
  \uses{lem:equalize, prereq:pair_cost_anti}
  For arms in an exponential family and a vector of means with unique best arm $a$, the optimal
  allocation is ordered like the means on the arms other than $a$: for $b, c \ne a$,
  $w^*_c(\mu) \le w^*_b(\mu)$ if and only if $\mu_c \le \mu_b$, and $w^*_c(\mu) < w^*_b(\mu)$ if
  and only if $\mu_c < \mu_b$ \cite[Proposition 6]{GK16}. The same holds for the $\beta$-optimal
  allocations, the maximizers of $D(\mu, w, \neg a)$ among the allocations with $w_a = \beta$: for
  $\beta > 0$, and in the strict form for $\beta \in (0, 1)$. No LMLRoadmaps target covers it yet.
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:equalize, prereq:pair_cost_anti}
  Let $w$ be $\beta$-optimal, so that the costs $C_b(w_b)$ of the pairs $(a, b)$, with
  $C_b(x) = \beta d(\mu_a, m_b) + x d(\mu_b, m_b)$, are all equal (\cref{lem:equalize} and its
  $\beta$ form). Let $b, c \ne a$ with $\mu_c \le \mu_b$. If $w_b < w_c$, then $C_b$ increases, so
  $C_b(w_b) < C_b(w_c)$. By \cref{prereq:pair_cost_anti}, $C_b(w_c) \le C_c(w_c)$. This
  contradicts $C_b(w_b) = C_c(w_c)$. If $\mu_c < \mu_b$ and $w_b \le w_c$, then
  $C_b(w_b) \le C_b(w_c) < C_c(w_c)$, using $w_c > 0$, a contradiction again. The optimal
  allocation is $\beta$-optimal for $\beta = w^*_a \in (0, 1)$; the two implications give the
  equivalences.
\end{proof}

\begin{lemma}
  \label{prereq:kt}
  \lean{LMLExplore.integral_pow_mul_one_sub_pow_betaMeasure,
    LMLExplore.div_pow_mul_div_pow_le_two_mul_sqrt_mul_integral_betaMeasure,
    LMLExplore.pow_mul_one_sub_pow_le_two_mul_sqrt_mul_integral_betaMeasure,
    LMLExplore.mul_klBerReal_div_le_log_two_mul_sqrt_mul_integral_betaMeasure_div}
  \leanok
  For a sequence of $n = k + m \ge 1$ observations with $k$ ones and $m$ zeros, let
  $\mathrm{kt}(k, m) = \int_0^1 u^k (1-u)^m \frac{du}{\pi\sqrt{u(1-u)}}
  = B(\tfrac12 + k, \tfrac12 + m) / B(\tfrac12, \tfrac12)$ be its Krichevsky--Trofimov
  probability, the mixture of the Bernoulli likelihoods over the arcsine law
  $\mathrm{Beta}(\tfrac12, \tfrac12)$. Then $u^k (1-u)^m \le (k/n)^k (m/n)^m \le 2\sqrt n\,
  \mathrm{kt}(k, m)$ for every $u \in [0, 1]$ \cite[Lemma 11]{GK16}, and for $p \in (0, 1)$,
  $n\, d(k/n, p) \le \log\bigl(2\sqrt n\, \mathrm{kt}(k, m) / (p^k (1-p)^m)\bigr)$. That
  $\mathrm{kt}$ is a probability on $\{0, 1\}^n$ is not formalized: the proof of
  \cref{prereq:kt_pair} does not need it.
\end{lemma}

\begin{proof}
  \leanok
  The moments of the beta laws give the closed form, and the recursion of the beta function
  $\mathrm{kt}(k, m + 1) = \mathrm{kt}(k, m)\,(m + \tfrac12)/(n + 1)$. By induction on $m$
  (from $\mathrm{kt}(0, 1) = \tfrac12$ and the symmetry in $k$ and $m$), the claim
  $k^k m^m \le 2\sqrt n\, n^n\, \mathrm{kt}(k, m)$ propagates from $m$ to $m + 1$ when
  $(m+1)^{m+1}\sqrt n\, n^n \le \sqrt{n+1}\,(m + \tfrac12)\, m^m (n+1)^n$, which in logarithms is
  $(m+1)\log(m+1) - m\log m - \log(m + \tfrac12) \le 1 \le (n + \tfrac12)\log(1 + 1/n)$. The
  right inequality is the first term of the series of $\log(1 + 1/n)$ in powers of $1/(2n+1)$;
  the left one, for $m \ge 1$, says that $t \mapsto (1+t)\log(1+t) - t - t\log(1 + t/2)$, whose
  derivative is $\log\frac{1+t}{1+t/2} - \frac{t}{2+t} \le 0$, is nonpositive at $t = 1/m$
  \cite[Appendix]{WST95}. The maximum likelihood bound is $n\, d(k/n, u) \ge 0$, and
  $n\, d(k/n, p) = \log\bigl((k/n)^k (m/n)^m / (p^k (1-p)^m)\bigr)$.
\end{proof}

\begin{lemma}
  \label{prereq:lr_supermartingale}
  \lean{LMLExplore.supermartingale_exp_sum_of_lintegral_exp_le_one,
    LMLExplore.measure_exists_le_integral_exp_sum_le}
  \leanok
  Along a run of any algorithm against a stochastic bandit $\nu$, if
  $\int e^{z(a, y)}\, d\nu_a(y) \le 1$ for every arm $a$, then
  $n \mapsto \exp\bigl(\sum_{k < n} z(A_k, Y_k)\bigr)$ is a supermartingale for the filtration of
  the history and the next action. If $z = z_\ell$ depends measurably on a parameter $\ell$ drawn
  from a probability measure $\rho$, then for every $c > 0$,
  $\Prob\bigl(\exists n,\ c \le \int \exp(\sum_{k<n} z_\ell(A_k, Y_k))\, d\rho(\ell)\bigr) \le 1/c$.
\end{lemma}

\begin{proof}
  \leanok
  Given the history and the action of round $n$, the observation $Y_n$ has law $\nu_{A_n}$, so
  the conditional expectation of $e^{z(A_n, Y_n)}$ is at most $1$. Mixtures of nonnegative
  supermartingales are supermartingales (in LMLExplore), with initial value $1$ here, and Ville's
  inequality (in LMLExplore) concludes.
\end{proof}

\begin{lemma}
  \label{prereq:kt_pair}
  \lean{LMLExplore.sum_ite_mul_add_one_sub_mul,
    LMLExplore.measure_exists_log_two_mul_mul_lt_klBer_le}
  \leanok
  \uses{prereq:kt, prereq:lr_supermartingale}
  Along a run of any algorithm against a Bernoulli bandit with means $p \in (0, 1)^K$, for two arms
  $a \ne b$ and every $x > 0$,
  $\Prob\bigl(\exists n,\ \log(2nx) < N_a(n)\, d(\hat\mu_a(n), p_a) + N_b(n)\, d(\hat\mu_b(n),
  p_b)\bigr) \le 1/x$ \cite[proof of Theorem 10]{GK16}.
\end{lemma}

\begin{proof}
  \leanok
  \uses{prereq:kt, prereq:lr_supermartingale}
  Take $z_{(u, v)}(c, y)$ the log-likelihood ratio of the observation $y$ of the arm $c$ under the
  bandit whose means on $a$ and $b$ are $u$ and $v$ against the true one, and
  $\rho = \mathrm{Beta}(\tfrac12, \tfrac12)^{\otimes 2}$. Its sum over the rounds is
  $S_a \log(u/p_a) + (N_a - S_a)\log((1-u)/(1-p_a))$ plus the same for $b$, so the mixture is the
  product $\Lambda_a \Lambda_b$ of the Krichevsky--Trofimov likelihood ratios
  $\Lambda_c = \mathrm{kt}(S_c, N_c - S_c) / (p_c^{S_c} (1-p_c)^{N_c - S_c})$, the observations
  being in $\{0, 1\}$ almost surely. By \cref{prereq:kt}, $N_c\, d(\hat\mu_c, p_c) \le
  \log(w_c \Lambda_c)$ with $w_c = 2\sqrt{N_c}$, or $w_c = 1$ when $N_c = 0$. On the event, one of
  the two arms has an observation (the right side is positive), and then $w_a w_b \le 2n$, so
  $\log(2nx) < \log(2n\Lambda_a\Lambda_b)$ and $x < \Lambda_a \Lambda_b$:
  \cref{prereq:lr_supermartingale} with $c = x$ concludes. The paper applies Lemma 11 also to an
  arm with no observation, where it fails (\texttt{notes/paper-errata.md}, E4).
\end{proof}

\begin{lemma}
  \label{prereq:kt_glr}
  \lean{LMLExplore.Identification.glr_plugInCost_baiTask_bernoulli_le}
  \leanok
  For best arm identification with Bernoulli arms on the vectors of means in $(0, 1)$, the GLR
  statistic of the plug-in statistic of the empirical means for the arm $i$ is at most
  $N_i\, d(\hat\mu_i, \lambda_i) + N_j\, d(\hat\mu_j, \lambda_j)$ for every arm $j \ne i$ and all
  $\lambda_i \le \lambda_j$ in $(0, 1)$.
\end{lemma}

\begin{proof}
  \leanok
  The cost of the alternatives to $i$ is the infimum of the transportation cost from the empirical
  means over the vectors $\lambda'$ of $(0, 1)^K$ with $\lambda'_i \le \lambda'_b$ for some
  $b \ne i$, since the cost is continuous there (in LMLExplore). Take $\lambda'$ with
  $\lambda'_i = \lambda_i$, $\lambda'_j = \lambda_j$, and
  $\lambda'_c = (1 - t)\hat p_c + t/2$ on the other arms, where $\hat p_c$ is $\hat\mu_c$ clamped to
  $[0, 1]$: by the $\chi^2$ bound, $d(\hat p_c, \lambda'_c) \le t$, and $t \to 0$.
\end{proof}

\begin{lemma}
  \label{prereq:kt_threshold}
  \lean{LMLExplore.Identification.measureReal_histOutputMeasure_glrt_baiTask_bernoulli_le}
  \leanok
  \uses{prereq:kt_pair, prereq:kt_glr}
  For best arm identification with Bernoulli arms on the vectors of means in $(0, 1)$ and every
  $\delta > 0$, whatever the sampling and recommendation rules, the GLRT with the plug-in statistic
  of the empirical means and thresholds at least $\log(2n(K-1)/\delta)$ stops with an arm that is
  not the best with probability at most $\delta$ \cite[Theorem 10]{GK16}. LMLRoadmaps:
  Identification 4.2.
\end{lemma}

\begin{proof}
  \leanok
  \uses{prereq:kt_pair, prereq:kt_glr}
  If the GLRT stops at round $n$ with an arm $i \ne a^*$, then by \cref{prereq:kt_glr} with
  $\lambda = \mu$ on the pair $(i, a^*)$,
  $\log(2n(K-1)/\delta) < N_i\, d(\hat\mu_i, \mu_i) + N_{a^*}\, d(\hat\mu_{a^*}, \mu_{a^*})$.
  By \cref{prereq:kt_pair} with $x = (K-1)/\delta$, each of the $K - 1$ wrong arms contributes at
  most $\delta/(K-1)$.
\end{proof}

\begin{lemma}
  \label{prereq:iid_slice}
  \lean{ProbabilityTheory.iIndepFun.measure_exists_le_and_gt_and_lt_rateFunction_le,
    ProbabilityTheory.iIndepFun.measure_exists_le_and_lt_and_le_mul_rateFunction_le}
  \leanok
  Let $Z_0, Z_1, \dots$ be i.i.d.\ with the law of an integrable $Y$, with rate function $I$, and
  let $\hat\mu_n$ be the mean of $Z_0, \dots, Z_{n-1}$. For $a > 0$ and $r \ge 0$,
  $\Prob(\exists n \ge a,\ \hat\mu_n > \E[Y],\ I(\hat\mu_n) > r) \le e^{-ar}$. For $a > 0$ and
  $z > 0$, $\Prob(\exists n,\ a \le n < b,\ n I(\hat\mu_n) \ge z) \le 2 e^{-az/b}$.
\end{lemma}

\begin{proof}
  \leanok
  Along the filtration of the past variables, $Z_n$ is independent of the past, so its
  conditional moment generating function is that of $Y$, and the first bound is LMLExplore's
  time-uniform Chernoff bound with the rate function, with every variable selected. For the
  second, $n I(\hat\mu_n) \ge z$ with $n < b$ forces $I(\hat\mu_n) > z/b$, which vanishes at the
  mean; above the mean this is the first bound with $r = z/b$, and below the mean the first bound
  for $-Z$ and $-Y$.
\end{proof}

\begin{lemma}
  \label{prereq:peeled_arm}
  \lean{LMLExplore.streamMeasure_exists_two_mul_add_le_mul_rateFunction_le}
  \leanok
  \uses{prereq:iid_slice}
  In the reward stream model, for an arm with integrable rewards and rate function $I_a$, every
  $\eta > 0$ and $y \ge 0$, with $\kappa_\eta = \log(2/(1 - e^{-\eta}))$,
  $\Prob(\exists s,\ s I_a(\hat\mu_s) \ge 2(\eta \lfloor\log_2 s\rfloor + \kappa_\eta + y))
  \le e^{-y}$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{prereq:iid_slice}
  A deviation with $s \ge 1$ rewards lies on the dyadic slice $2^k \le s < 2^{k+1}$ with
  $k = \lfloor\log_2 s\rfloor$, and at $s = 0$ there is none. By \cref{prereq:iid_slice} with
  $a = 2^k$ and $b = 2^{k+1}$, the slice $k$ has probability at most
  $2 e^{-(\eta k + \kappa_\eta + y)}$, and these sum to
  $2 e^{-\kappa_\eta - y} / (1 - e^{-\eta}) = e^{-y}$.
\end{proof}

\begin{lemma}
  \label{prereq:peeled_joint}
  \lean{LMLExplore.streamMeasure_exists_two_mul_card_mul_add_le_sum_mul_rateFunction_le,
    LMLExplore.measure_exists_two_mul_card_mul_add_le_sum_pullCount_mul_rateFunction_le,
    LMLExplore.measure_exists_two_mul_card_mul_add_le_sum_pullCount_mul_expFamilyRate_le}
  \leanok
  \uses{prereq:peeled_arm}
  Along a run of any algorithm, if the rewards of each arm are integrable with rate function
  $I_a$, then for $\eta > 0$ and $x \ge K$,
  $\Prob(\exists t,\ \sum_a N_a(t) I_a(\hat\mu_a(t)) \ge
  2(K(\eta \lfloor\log_2 t\rfloor + \kappa_\eta) + x)) \le (x/K)^K e^{K - x}$. For arms dominated
  by members of exponential families, the same holds with the extended divergences
  $\bar d_a(\hat\mu_a(t), \mu_a)$ in place of $I_a(\hat\mu_a(t))$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{prereq:peeled_arm}
  In the stream model the arms are independent, and \cref{prereq:peeled_arm} gives each half
  statistic $s I_a(\hat\mu_s)/2$ the exponential tail above the monotone function
  $\eta \lfloor\log_2 s\rfloor + \kappa_\eta$ that \cite[Proposition 33.9]{LS} (in LMLExplore)
  combines over the arms. LMLExplore's joint event transfer turns the stream bound into a bound
  along the run, where $\sum_a N_a(t) = t$. The rate function of a dominated arm is at least the
  extended divergence of its family.
\end{proof}

\begin{lemma}
  \label{prereq:deviational_glrt}
  \lean{LMLExplore.Identification.IdentTask.measure_exists_lt_expFamilyEvidence_le_of_le,
    LMLExplore.Identification.IdentTask.measure_exists_lt_expFamilyEvidence_le_of_mul_log_le,
    LMLExplore.Identification.IdentTask.measureReal_histOutputMeasure_glrt_expFamilyEvidence_le_of_mul_log_le}
  \leanok
  \uses{prereq:peeled_joint}
  For every identification task on vectors of means, arms dominated by exponential families,
  every sampling rule, $p > 0$ and $x \ge K$, the GLRT with the generalized likelihood ratio
  evidence and thresholds at least $p \log n + c_{p,K} + 2x$ at every round $n \ge 1$, with
  $c_{p,K} = 2K \log(2/(1 - e^{-p\log 2/(2K)}))$, stops with an incorrect answer with
  probability at most $(x/K)^K e^{K-x}$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{prereq:peeled_joint}
  The GLRT errs only if the evidence against the true model exceeds the threshold at some round,
  which does not happen at round $0$, where the evidence vanishes. With
  $\eta = p\log 2/(2K)$, $2K\eta\lfloor\log_2 n\rfloor \le p \log n$, so the thresholds dominate
  those of \cref{prereq:peeled_joint}.
\end{proof}

\begin{lemma}

  \lean{LMLExplore.Identification.isOpen_answerSet_baiTask_restrict,
    LMLExplore.Identification.continuousWithinAt_altCost_baiTask_restrict}
  \leanok
  For best arm identification restricted to an open set of vectors of means, such as the vectors
  of means of a nondegenerate exponential family, the models with unique best arm $a$ form an open
  set. For arms in a nondegenerate exponential family and best arm identification restricted to
  the vectors of means of the family, at every model $\mu$ with unique best arm $a$,
  $(\mu', w) \mapsto D(\mu', w, \neg a)$ is continuous within $\R^K \times \Sk$ at $(\mu, w)$ for

\end{lemma}

\begin{proof}
  \leanok
  The models with unique best arm $a$ are an intersection of finitely many strict inequalities
  between means, intersected with the open set of the restriction. The closed form of
  \cref{lem:pair_cost} is continuous on the vectors of means of the family with best arm $a$ and

\end{proof}

\begin{lemma}
  \label{prereq:as_pathwise}
  \lean{LMLExplore.Identification.IdentTask.stoppingTime_glrt_ne_top_of_eventually_mul_le_glr,
    LMLExplore.Identification.IdentTask.eventually_stoppingTime_glrt_le_of_eventually_mul_le_glr}
  \leanok
  Along a sequence of histories on which the GLR statistic of the recommended answer after $t$
  rounds is at least $\kappa t$ for all large $t$, consider the GLRTs with thresholds
  $\beta_t(h, \delta) \le \log(1/\delta) + f(t)$, $f(t) = o(t)$. If $\kappa > 0$, the GLRT stops
  at every $\delta$ at which the thresholds have this form, and $\tau_\delta \le c \log(1/\delta)$
  for all small $\delta$, for every $c > 1/\kappa$.
\end{lemma}

\begin{proof}
  \leanok
  Take $1/c < \eta < \kappa$ and $L = \log(1/\delta)$. From the round $N = \max(M, \lceil L/\eta
  \rceil)$ on, for a fixed $M$, the threshold is at most $L + (\kappa - \eta) N / 2 < \kappa N$, so
  that $\tau_\delta \le N \le M + L/\eta + 1$, which is at most $cL$ when $L$ is large.
\end{proof}

\begin{lemma}
  \label{prereq:glr_growth}
  \lean{LMLExplore.Identification.IdentTask.eventually_mul_le_glr_plugInCost}
  \leanok
  Let $W$ be a compact set of allocations such that $(\theta', w) \mapsto D(\theta', w, \neg i)$ is
  lower semicontinuous at $(\theta, w)$ on the simplex for every $w \in W$, and let the
  recommendation be $i$ near $\theta$. Along a sequence of histories on which the estimates
  converge to $\theta$ and the proportions of pulls $N(t)/t$ approach $W$, the GLR statistic of the
  plug-in statistic, $t D(\hat\theta_t, N(t)/t, \neg i)$, is at least $\kappa t$ for all large $t$,
  for every $\kappa < \inf_{w \in W} D(\theta, w, \neg i)$.
\end{lemma}

\begin{proof}
  \leanok
  By compactness of $W$ and lower semicontinuity, $D(\theta', w', \neg i) > \kappa$ for $\theta'$
  near $\theta$ and $w'$ in the simplex near $W$, which $\hat\theta_t$ and $N(t)/t$ are for all
  large $t$.
\end{proof}

\begin{lemma}
  \label{prereq:as}
  \lean{LMLExplore.Identification.IdentTask.ae_eventually_stoppingTime_glrt_le,
    LMLExplore.Identification.IdentTask.ae_stoppingTime_glrt_ne_top}
  \leanok

  allocations, then almost surely $\tau_\delta \le c \log(1/\delta)$ for all small $\delta$, for

  this form. LMLRoadmaps: Identification 4.4.

\begin{proof}

\end{proof}

\begin{lemma}
  \label{prereq:slln}
  \lean{LMLExplore.ae_tendsto_empMean_of_tendsto_pullCount}
  \leanok
  In a run of any algorithm against a stochastic bandit with integrable rewards, almost surely the
  empirical mean of every arm pulled infinitely often converges to the mean of the arm.
\end{lemma}

\begin{proof}
  \leanok
  The rewards of the successive pulls of an arm are i.i.d.\ with the law of the arm (in LML), and
  after $N_a(t)$ pulls the empirical mean is the mean of the first $N_a(t)$ of them: the strong law
  of large numbers applies.
\end{proof}
